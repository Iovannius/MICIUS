cmake_minimum_required(VERSION 3.21)
project(MiciusFileConsumer LANGUAGES C CXX)
set(PARSERS "both" CACHE STRING "llama, qwen2 or both")
set(components core gguf)
if(PARSERS MATCHES "^(both|llama)$")
    list(APPEND components parser_llama)
endif()
if(PARSERS MATCHES "^(both|qwen2)$")
    list(APPEND components parser_qwen2)
endif()
if(NOT PARSERS MATCHES "^(both|llama|qwen2)$")
    message(FATAL_ERROR "Select llama, qwen2 or both")
endif()
find_package(MICIUS 0.7.1 CONFIG REQUIRED COMPONENTS ${components})
add_executable(micius_model_probe model_probe.c)
set_target_properties(micius_model_probe PROPERTIES LINKER_LANGUAGE CXX)
foreach(component IN LISTS components)
    target_link_libraries(micius_model_probe PRIVATE MICIUS::${component})
endforeach()
if("parser_llama" IN_LIST components)
    target_compile_definitions(micius_model_probe PRIVATE LINK_LLAMA)
endif()
if("parser_qwen2" IN_LIST components)
    target_compile_definitions(micius_model_probe PRIVATE LINK_QWEN2)
endif()

if(DEFINED V07_CLIENT_OBJECT)
    set_source_files_properties("${V07_CLIENT_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE)
    add_executable(v07_client "${V07_CLIENT_OBJECT}")
    set_target_properties(v07_client PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(v07_client PRIVATE MICIUS::core)
endif()
